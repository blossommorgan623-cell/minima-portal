// ==UserScript==
// @name         Zealy Auto-Claim
// @namespace    http://tampermonkey.net/
// @version      3.1
// @match        https://zealy.io/*
// @grant        none
// ==/UserScript==
(function () {
    setInterval(() => {
        const sel = '[role="radio"], [role="option"], input[type="radio"], div[class*="option"]';
        document.querySelectorAll(sel).forEach(el => {
            if (el.offsetParent && !el.disabled && el.getAttribute('aria-checked') !== 'true' && !el.checked) el.click();
        });
        const btns = [...document.querySelectorAll('button')];
        const btn = btns.find(b => !b.disabled && b.offsetParent && ['claim','claim reward','claim quest','next','submit'].some(t => b.innerText.toLowerCase().includes(t)));
        if (btn) btn.click();
    }, 50);
})();

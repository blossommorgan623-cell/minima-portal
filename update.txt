// ==UserScript==
// @name         Zealy Fast Auto-Claim (PC)
// @namespace    http://tampermonkey.net/
// @version      1.3
// @description  Fast auto-claim helper for Zealy on PC
// @match        https://zealy.io/*
// @grant        none
// ==/UserScript==

(function () {
    'use strict';

    let clicking = false;

    function findClaimButton() {
        const buttons = [...document.querySelectorAll('button')];

        return buttons.find(btn => {
            const text = btn.innerText
                .trim()
                .replace(/\s+/g, ' ')
                .toLowerCase();

            return (
                (text === 'claim' || text === 'claim reward' || text === 'claim quest') &&
                !btn.disabled &&
                btn.offsetParent !== null
            );
        });
    }

    function autoClaim() {
        if (clicking) return;

        const btn = findClaimButton();

        if (btn) {
            clicking = true;
            btn.click();
            console.log('[Zealy PC Auto-Claim] Claimed');

            setTimeout(() => {
                clicking = false;
            }, 300);
        }
    }

    // Observer for fast dynamic page changes
    const observer = new MutationObserver(autoClaim);
    observer.observe(document.body, {
        childList: true,
        subtree: true
    });

    // Backup interval polling
    setInterval(autoClaim, 200);
})();
